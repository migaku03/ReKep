def stage3_subgoal_constraint1(end_effector, keypoints):
    """The white pen must be 5cm above the black pen holder (keypoint 2)."""
    target_point = keypoints[2] + np.array([0, 0, 5])  # 5cm above the pen holder
    cost = np.linalg.norm(end_effector - target_point)  # Cost is the distance to the target drop position
    return cost
