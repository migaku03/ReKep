def stage3_subgoal_constraint1(end_effector, keypoints):
    """Ensure the white pen (keypoints 0 and 1) is upright (parallel to the z-axis)."""
    pen_vector = keypoints[1] - keypoints[0]
    z_axis = np.array([0, 0, 1])
    cost = 1 - np.dot(pen_vector / np.linalg.norm(pen_vector), z_axis)
    return cost

def stage3_subgoal_constraint2(end_effector, keypoints):
    """Ensure the white pen's bottom (keypoint 0) is 5cm above the black pen holder's opening (mean of keypoints 2 and 3)."""
    # HAND-CORRECTED, the one deliberate edit in this directory. gpt-4o-2024-11-20 wrote
    # keypoints[2] alone, but keypoints 2 and 3 sit on opposite sides of the rim, 14.0cm apart,
    # so a single one of them is 6.9cm off the centre -- about the radius of the opening, i.e.
    # aimed at the wall. output_raw.txt is left untouched as the record of what the model said.
    holder_opening = np.mean(keypoints[2:4], axis=0)
    offsetted_point = holder_opening + np.array([0, 0, 0.05])
    cost = np.linalg.norm(keypoints[0] - offsetted_point)
    return cost
